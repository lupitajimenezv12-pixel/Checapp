function traducir(t){
  if(/Invalid login/i.test(t))return 'Correo o contraseña incorrectos';
  if(/already registered/i.test(t))return 'Ese correo ya tiene una cuenta';
  if(/at least/i.test(t))return 'La contraseña debe tener al menos 6 caracteres';
  if(/rate limit/i.test(t))return 'Demasiados intentos, espera un momento';
  return t;
}
function mostrar(v){$('auth').hidden=v!=='auth';$('home').hidden=v!=='home';$('salir').hidden=v!=='home'}
function hora(f){return new Date(f).toLocaleTimeString('es-MX',{hour:'2-digit',minute:'2-digit'})}

function setModo(m){
  modo=m;
  $('tabE').classList.toggle('on',m==='entrar');
  $('tabC').classList.toggle('on',m==='crear');

  e.preventDefault();msg('');

  try{
    if(modo==='entrar'){
      const {error}=await SB.auth.signInWithPassword({email,password:pass});

async function iniciar(){
  const {data:{session}}=await SB.auth.getSession();
  if(!session){mostrar('auth');return}
  const {data,error}=await SB.from('perfiles').select('id,nombre,rol,empresa_id,empresas(nombre)').eq('id',session.user.id).maybeSingle();

  $('hola').textContent='Hola, '+data.nombre.split(' ')[0];
  $('emp').textContent=data.empresas.nombre;
  msg('');mostrar('home');
  await refrescar();
}

async function refrescar(){
  const ini=new Date();ini.setHours(0,0,0,0);
  const {data}=await SB.from('registros').select('tipo,fecha_hora,es_retardo').eq('empleado_id',perfil.id).gte('fecha_hora',ini.toISOString()).order('fecha_hora',{ascending:false});
  const l=data||[];
  tipoSig=(!l.length||l[0].tipo==='salida')?'entrada':'salida';
  $('btnReg').textContent=tipoSig==='entrada'?'Registrar entrada':'Registrar salida';
  $('btnReg').className='reg '+tipoSig;
  $('lista').innerHTML=l.length
    ?l.map(r=>'<li><span>'+(r.tipo==='entrada'?'Entrada':'Salida')+'</span><span>'+hora(r.fecha_hora)+(r.es_retardo?' <b class="tarde">Retardo</b>':'')+'</span></li>').join('')
    :'<li><span>Aún no hay registros hoy</span></li>';
}

function ubicacion(){
  return new Promise(ok=>{
    if(!navigator.geolocation)return ok(null);
    navigator.geolocation.getCurrentPosition(p=>ok(p.coords),()=>ok(null),{enableHighAccuracy:true,timeout:10000});
  });
}
function comprimir(f){
  return new Promise((ok,no)=>{
    const i=new Image();
    i.onload=()=>{
      const s=Math.min(1,800/Math.max(i.width,i.height));
      const c=document.createElement('canvas');
      c.width=Math.round(i.width*s);c.height=Math.round(i.height*s);
      c.getContext('2d').drawImage(i,0,0,c.width,c.height);
      c.toBlob(b=>b?ok(b):no(new Error('Foto inválida')),'image/jpeg',0.7);
    };
    i.onerror=()=>no(new Error('Foto inválida'));
    i.src=URL.createObjectURL(f);
  });
}

    await refrescar();
  }catch(err){msg('No se pudo registrar: '+err.message)}
  b.disabled=false;
};
